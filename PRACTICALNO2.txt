#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

void ascendingSort(int a[], int n)
{
    int i, j, temp;

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
}

void descendingSort(int a[], int n)
{
    int i, j, temp;

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] < a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
}

int main()
{
    int a[20], n, i;
    pid_t pid;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if(n < 1 || n > 20)
    {
        printf("Invalid number of elements!\n");
        return 1;
    }

    printf("Enter elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    pid = fork();

    if(pid == 0)
    {
        printf("\nChild Process (Ascending Order)\n");

        ascendingSort(a, n);

        printf("Sorted array: ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        printf("\n");
    }
    else if(pid > 0)
    {
        wait(NULL);

        printf("\nParent Process (Descending Order)\n");

        descendingSort(a, n);

        printf("Sorted array: ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        printf("\n");
    }
    else
    {
        printf("Fork failed!\n");
        return 1;
    }

    return 0;
}
